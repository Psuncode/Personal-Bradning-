Create `src/app/actions/contact.ts` with the following exact content:

```typescript
'use server';

import { db } from '@/db';
import { contacts } from '@/db/schema';
import { headers } from 'next/headers';

export type ContactFormState = {
  success: boolean;
  error?: string;
};

const initialState: ContactFormState = { success: false };

export async function saveContact(
  _prev: ContactFormState,
  formData: FormData
): Promise<ContactFormState> {
  const name = (formData.get('name') as string)?.trim();
  const email = (formData.get('email') as string)?.trim();
  const subject = (formData.get('subject') as string)?.trim();
  const message = (formData.get('message') as string)?.trim();

  if (!name || !email || !message) {
    return { success: false, error: 'Name, email, and message are required.' };
  }

  const headersList = await headers();
  const referer = headersList.get('referer') ?? null;

  try {
    await db.insert(contacts).values({
      name,
      email,
      subject: subject || null,
      message,
      utmSource: (formData.get('utm_source') as string) || null,
      utmMedium: (formData.get('utm_medium') as string) || null,
      utmCampaign: (formData.get('utm_campaign') as string) || null,
      referrer: referer,
    });
    return { success: true };
  } catch (err) {
    console.error('saveContact error:', err);
    return { success: false, error: 'Something went wrong. Please try again.' };
  }
}
```

Key implementation details:
- `'use server'` directive at the top makes this a Server Action
- Signature `(_prev: ContactFormState, formData: FormData)` is the `useActionState` pattern (React 19) — first arg is previous state, second is form data
- UTM params read from `formData` (client component passes them as hidden inputs named `utm_source`, `utm_medium`, `utm_campaign`)
- HTTP `Referer` header read from `headers()` (the page the user came from)
- Returns `{ success: boolean, error?: string }` — client uses this to show success/error states
- Empty strings coerced to `null` for UTM fields (avoids storing empty strings in DB)
